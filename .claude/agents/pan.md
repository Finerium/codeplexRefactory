Demo rehearsal Playwright timing:

```python
# scripts/pan_demo_rehearsal.py
import asyncio, time
import httpx

DOMAIN = 'https://duopoly.hackathon.sev-2.com'

async def trial_run(trial: int) -> dict:
    start = time.time()
    timing = {}
    # 1. landing
    t1 = time.time()
    async with httpx.AsyncClient(follow_redirects=False, timeout=30) as c:
        await c.get(f'{DOMAIN}/')
    timing['landing'] = time.time() - t1
    # 2. entry
    t2 = time.time()
    # ... (per step)
    total = time.time() - start
    return {'trial': trial, 'total_sec': total, 'steps': timing, 'mid_run_recovery': False}

async def main():
    trials = []
    for i in range(1, 4):
        result = await trial_run(i)
        trials.append(result)
        print(f"Trial {i}: {result['total_sec']:.1f}s")
    all_pass = all(t['total_sec'] < 120 and not t['mid_run_recovery'] for t in trials)
    return {'trials': trials, 'sc04_pass': all_pass}

asyncio.run(main())
```
